# PsocDuino - Arduino core para Cypress PSoC 1 (M8C)
# https://github.com/CEPH81/PsocDuino

name=PsocDuino (Cypress PSoC 1)
# Publicar en el Gestor de placas: cambie la version y haga push a main (release.yml)
version=0.2.0

# ---------------------------------------------------------------------------
# Compilacion: ImageCraft ICCM8C de PSoC Designer 5.x (solo Windows), a traves
# de psocbuild, que adapta los .ino/.cpp (ImageCraft solo compila C), usa
# rutas cortas, convierte los errores al formato del IDE y enlaza con la
# configuracion de la variante (importada de un proyecto de PSoC Designer).
# Si PSoC Designer no esta en la ruta por defecto, defina la variable de
# entorno PSOC_DESIGNER_DIR=...\PSoC Designer\5.4\Common\CypressSemiBuildMgr
# ---------------------------------------------------------------------------
psocbuild.path={runtime.platform.path}/tools/bin
psocbuild.path.windows={runtime.platform.path}/tools/bin/windows
psocbuild.path.linux={runtime.platform.path}/tools/bin/linux
psocbuild.path.macosx={runtime.platform.path}/tools/bin/macos
psocbuild.cmd=psocbuild
psocbuild.cmd.windows=psocbuild.exe
psocbuild="{psocbuild.path}/{psocbuild.cmd}"

compiler.defines=-DARDUINO={runtime.ide.version} -DARDUINO_{build.board} -DARDUINO_ARCH_M8C -DPSOC_NUM_PORTS={build.ports}
compiler.c.extra_flags=
compiler.cpp.extra_flags=
build.extra_flags=

recipe.c.o.pattern={psocbuild} cc -V "{build.variant.path}" -o "{object_file}" "{source_file}" {compiler.defines} {compiler.c.extra_flags} {build.extra_flags} {includes}
recipe.cpp.o.pattern={psocbuild} cc -V "{build.variant.path}" -o "{object_file}" "{source_file}" {compiler.defines} {compiler.cpp.extra_flags} {build.extra_flags} {includes}
recipe.S.o.pattern={psocbuild} as -V "{build.variant.path}" -o "{object_file}" "{source_file}" {includes}
recipe.ar.pattern={psocbuild} ar -V "{build.variant.path}" "{archive_file_path}" "{object_file}"
recipe.c.combine.pattern={psocbuild} link -V "{build.variant.path}" -o "{build.path}/{build.project_name}" {object_files} "{archive_file_path}"
recipe.preproc.macros={psocbuild} preproc "{source_file}" -o "{preprocessed_file_path}"

recipe.size.pattern={psocbuild} size "{build.path}/{build.project_name}.mp"
recipe.size.regex=^Flash:\s+([0-9]+).*
recipe.size.regex.data=^RAM:\s+([0-9]+).*

# ---------------------------------------------------------------------------
# Programador: psocprog + CH341A
# ---------------------------------------------------------------------------
tools.psocprog.path={runtime.platform.path}/tools/bin
tools.psocprog.path.windows={runtime.platform.path}/tools/bin/windows
tools.psocprog.path.linux={runtime.platform.path}/tools/bin/linux
tools.psocprog.path.macosx={runtime.platform.path}/tools/bin/macos
tools.psocprog.cmd=psocprog
tools.psocprog.cmd.windows=psocprog.exe

tools.psocprog.upload.params.verbose=-v
tools.psocprog.upload.params.quiet=
tools.psocprog.upload.pattern="{path}/{cmd}" {upload.verbose} -c {upload.chip} -m {upload.mode} -V {upload.vdd} --verify {upload.psoc_verify} {upload.secure_flag} --run -w "{build.path}/{build.project_name}.hex"

# "Subir usando programador" (mismo comando)
tools.psocprog.program.params.verbose=-v
tools.psocprog.program.params.quiet=
tools.psocprog.program.pattern="{path}/{cmd}" {program.verbose} -c {upload.chip} -m {upload.mode} -V {upload.vdd} --verify {upload.psoc_verify} {upload.secure_flag} --run -w "{build.path}/{build.project_name}.hex"

# "Quemar bootloader" = borrar el chip completo (quita tambien la proteccion)
tools.psocprog.erase.params.verbose=-v
tools.psocprog.erase.params.quiet=
tools.psocprog.erase.pattern="{path}/{cmd}" {erase.verbose} -c {upload.chip} -m {upload.mode} -V {upload.vdd} -e
tools.psocprog.bootloader.params.verbose=-v
tools.psocprog.bootloader.params.quiet=
tools.psocprog.bootloader.pattern="{path}/{cmd}" {bootloader.verbose} -c {upload.chip} -m {upload.mode} -V {upload.vdd} -i
